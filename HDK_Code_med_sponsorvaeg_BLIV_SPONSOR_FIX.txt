function pickStr_(obj, keys) {
  if (!obj) return "";
  for (var i = 0; i < keys.length; i++) {
    var k = keys[i];
    if (!k) continue;
    if (obj.hasOwnProperty(k) && obj[k] !== undefined && obj[k] !== null && String(obj[k]).trim() !== "") {
      return String(obj[k]).trim();
    }
  }
  return "";
}

/**
 * Send email med bedste muligt "klub-afsender"-setup.
 * - replyTo sættes til klubmail (så svar går til klubben)
 * - name sættes til "Humlum Dartklub"
 * - from forsøges sat til klubmail (kræver send-as/alias ELLER at script kører som klubkonto)
 *
 * Fejl i mail må aldrig stoppe API-responsen.
 */
function sendEmailSafe_(to, subject, body, options) {
  if (!to) return false;

  var clubEmail = getClubEmail_();
  var opts = options || {};

  // standarder
  if (!opts.name) opts.name = "Humlum Dartklub";
  if (clubEmail && !opts.replyTo) opts.replyTo = clubEmail;

  // 1) Prøv GmailApp med from (hvis muligt)
  try {
    var o1 = {};
    for (var k in opts) o1[k] = opts[k];
    if (clubEmail) o1.from = clubEmail; // kræver alias eller at script kører som klubkonto
    GmailApp.sendEmail(String(to).trim(), String(subject || "").trim(), String(body || ""), o1);
    return true;
  } catch (e1) {
    // 2) Prøv GmailApp uden from
    try {
      var o2 = {};
      for (var k2 in opts) o2[k2] = opts[k2];
      if (o2.from) delete o2.from;
      GmailApp.sendEmail(String(to).trim(), String(subject || "").trim(), String(body || ""), o2);
      return true;
    } catch (e2) {
      // 3) Fallback: MailApp helt uden options
      try {
        MailApp.sendEmail(String(to).trim(), String(subject || "").trim(), String(body || ""));
        return true;
      } catch (e3) {
        return false;
      }
    }
  }
}

function makeMemberId_(memberRowIndex) {
  var n = parseInt(memberRowIndex, 10);
  if (!n || n < 2) n = 2;
  var num = n - 1; // row 2 => 1
  return "HDK-" + ("0000" + num).slice(-4);
}

function makeMemberCode_() {
  // 6-cifret personlig kode til medlemslogin
  return String(Math.floor(100000 + Math.random() * 900000));
}

function getRowObject_(sheet, rowIndex) {
  var header = getHeaderMap(sheet);
  var headers = header.headers;
  var lastCol = header.lastCol;
  var row = sheet.getRange(rowIndex, 1, 1, lastCol).getValues()[0];
  var obj = {};
  for (var i = 0; i < headers.length; i++) {
    var key = headers[i];
    if (!key) continue;
    obj[key] = row[i];
  }
  return obj;
}

// ARCADE: træk vinder blandt alle købte forsøg (tickets)
function arcadeDraw_(e) {
  if (!arcadeKeyOk_(e)) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: "unauthorized" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const body = JSON.parse(e.postData && e.postData.contents ? e.postData.contents : "{}");
  const eventCode = String(body.event_code || "").trim();
  if (!eventCode) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: "Missing event_code" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const ss = SpreadsheetApp.openById(
    PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID")
  );
  const shEvents = ss.getSheetByName("ARCADE_EVENTS");
  const shPurch = ss.getSheetByName("ARCADE_PURCHASES");
  if (!shEvents) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: "Missing sheet tab ARCADE_EVENTS" }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  if (!shPurch) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: "Missing sheet tab ARCADE_PURCHASES" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const evValues = shEvents.getDataRange().getValues();
  const evHead = evValues[0] || [];
  const idxCode = evHead.indexOf("event_code");
  const idxWinner = evHead.indexOf("draw_winner_nickname");
  const idxNote = evHead.indexOf("draw_winner_note");
  const idxDrawnAt = evHead.indexOf("draw_drawn_at");
  const idxStatus = evHead.indexOf("draw_status");
  const idxSeed = evHead.indexOf("draw_seed");

  if (idxCode < 0) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: "ARCADE_EVENTS missing header event_code" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  let evRow = -1;
  for (let r = 1; r < evValues.length; r++) {
    if (String(evValues[r][idxCode] || "").trim() === eventCode) {
      evRow = r + 1; // 1-indexed sheet row
      break;
    }
  }
  if (evRow < 0) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: "Event not found: " + eventCode }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const pValues = shPurch.getDataRange().getValues();
  const pHead = pValues[0] || [];
  const pIdxEvent = pHead.indexOf("event_code");
  const pIdxNick = pHead.indexOf("nickname");
  const pIdxBought = pHead.indexOf("purchased_attempts");

  if (pIdxEvent < 0 || pIdxNick < 0 || pIdxBought < 0) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: "ARCADE_PURCHASES missing required headers" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  let total = 0;
  const buckets = []; // {nickname, count}
  for (let r = 1; r < pValues.length; r++) {
    if (String(pValues[r][pIdxEvent] || "").trim() !== eventCode) continue;
    const nick = String(pValues[r][pIdxNick] || "").trim() || "Anonym";
    const cnt = Math.max(1, Number(pValues[r][pIdxBought] || 1));
    total += cnt;
    buckets.push({ nickname: nick, count: cnt });
  }

  if (total <= 0) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: "No purchases for event" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // seed: hvis du vil have reproducérbart lodtrækning
  const seedStr = String(body.draw_seed || "").trim();
  let seed = seedStr ? seedStr : (new Date().toISOString());
  // enkel pseudo-random fra seed
  function hash_(s) {
    let h = 2166136261;
    for (let i = 0; i < s.length; i++) {
      h ^= s.charCodeAt(i);
      h = Math.imul(h, 16777619);
    }
    return (h >>> 0);
  }
  let rnd = hash_(seed) / 4294967296;
  const pick = Math.floor(rnd * total) + 1; // 1..total

  let acc = 0;
  let winner = "";
  for (let i = 0; i < buckets.length; i++) {
    acc += buckets[i].count;
    if (pick <= acc) {
      winner = buckets[i].nickname;
      break;
    }
  }

  const drawnAt = new Date().toISOString();
  const note = "Trukket blandt " + total + " købte forsøg · ticket #" + pick;

  if (idxWinner >= 0) shEvents.getRange(evRow, idxWinner + 1).setValue(winner);
  if (idxNote >= 0) shEvents.getRange(evRow, idxNote + 1).setValue(note);
  if (idxDrawnAt >= 0) shEvents.getRange(evRow, idxDrawnAt + 1).setValue(drawnAt);
  if (idxStatus >= 0) shEvents.getRange(evRow, idxStatus + 1).setValue("DRAWN");
  if (idxSeed >= 0) shEvents.getRange(evRow, idxSeed + 1).setValue(seed);

  return ContentService
    .createTextOutput(JSON.stringify({ ok: true, event_code: eventCode, total_tickets: total, ticket: pick, winner_nickname: winner, drawn_at: drawnAt, note: note, seed: seed }))
    .setMimeType(ContentService.MimeType.JSON);
}

function strv_(v) {
  return (v === undefined || v === null) ? "" : String(v).trim();
}

function boolYesNo_(v, fallback) {
  var s = strv_(v).toLowerCase();
  if (!s && fallback !== undefined) return boolYesNo_(fallback);

  if (s === "yes" || s === "ja" || s === "true" || s === "1") return "YES";
  if (s === "no" || s === "nej" || s === "false" || s === "0") return "NO";

  return s ? String(v).toUpperCase() : "";
}

function slugifyDa_(value) {
  var s = strv_(value).toLowerCase();
  s = s.replace(/æ/g, "ae").replace(/ø/g, "oe").replace(/å/g, "aa");
  s = s.replace(/[^a-z0-9]+/g, "-");
  s = s.replace(/^-+|-+$/g, "");
  return s;
}

function normalizeAlbumKey_(value) {
  return slugifyDa_(value).replace(/-/g, "_");
}

function albumKeyToTitle_(value) {
  var raw = strv_(value).toLowerCase();

  var map = {
    "traening": "Træning",
    "ungdom": "Ungdom",
    "turnering": "Turnering",
    "events": "Events",
    "sommerfest": "Sommerfest",
    "holdturnering": "Holdturnering",
    "aabningsdag": "Åbningsdag",
    "klubhus": "Klubhus",
    "andet": "Andet"
  };

  if (map[raw]) return map[raw];

  var s = strv_(value).replace(/[_-]+/g, " ");
  if (!s) return "";
  return s.replace(/\b\w/g, function(ch) { return ch.toUpperCase(); });
}

function headerIdx_(headers, name) {
  return headers.indexOf(name);
}

function getByHeader_(headers, rowValues, name) {
  var idx = headerIdx_(headers, name);
  return idx === -1 ? "" : rowValues[idx];
}

function setByHeader_(headers, rowValues, name, value) {
  var idx = headerIdx_(headers, name);
  if (idx !== -1) rowValues[idx] = value;
}

function isYesLike_(v) {
  var s = strv_(v).toLowerCase();
  return s === "yes" || s === "ja" || s === "true" || s === "1";
}

function applyGalleryRowDefaults_(headers, rowValues) {
  var albumKey = strv_(getByHeader_(headers, rowValues, "album_key"));
  var category = strv_(getByHeader_(headers, rowValues, "category"));
  var title = strv_(getByHeader_(headers, rowValues, "title"));
  var key = strv_(getByHeader_(headers, rowValues, "key"));
  var altText = strv_(getByHeader_(headers, rowValues, "alt_text"));
  var type = strv_(getByHeader_(headers, rowValues, "type"));
  var source = strv_(getByHeader_(headers, rowValues, "source"));
  var visible = getByHeader_(headers, rowValues, "visible");
  var order = strv_(getByHeader_(headers, rowValues, "order"));

  if (!albumKey) albumKey = normalizeAlbumKey_(category || title || key || "andet");
  if (!category) category = albumKeyToTitle_(albumKey) || "Andet";
  if (!title) title = "Galleri-billede";
  if (!altText) altText = title;
  if (!type) type = "image";
  if (!source) source = "admin";
  if (!strv_(visible)) visible = "YES";
  if (!order) order = String(new Date().getTime());

  setByHeader_(headers, rowValues, "album_key", albumKey);
  setByHeader_(headers, rowValues, "category", category);
  setByHeader_(headers, rowValues, "title", title);
  setByHeader_(headers, rowValues, "alt_text", altText);
  setByHeader_(headers, rowValues, "type", type);
  setByHeader_(headers, rowValues, "source", source);
  setByHeader_(headers, rowValues, "visible", boolYesNo_(visible, "YES"));
  setByHeader_(headers, rowValues, "order", order);
}

  var albumKey = strv_(getByHeader_(headers, rowValues, "album_key"));

  if (!order) order = "999";

  setByHeader_(headers, rowValues, "album_key", albumKey);
  setByHeader_(headers, rowValues, "title", title);
  setByHeader_(headers, rowValues, "slug", slug);
  setByHeader_(headers, rowValues, "visible", boolYesNo_(visible, "YES"));
  setByHeader_(headers, rowValues, "source", source);
  setByHeader_(headers, rowValues, "order", order);
}


/**************************************
 * SPONSORVÆG V2
 * - tilføjer felter til SPONSORER
 * - migrerer eksisterende data én gang

function sponsorWallSetup_() {
  var sheet = getSheetByName("SPONSORER");
  if (!sheet) throw new Error("Unknown tab: SPONSORER");

  var wanted = ["wall_title", "wall_subtitle", "wall_extra", "wall_show_extra", "wall_kind", "wall_category", "wall_lane", "wall_visible", "wall_order"];
  var lastCol = sheet.getLastColumn();
  var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(function(v) {
    return String(v || "").trim();
  });

  wanted.forEach(function(name) {
    if (headers.indexOf(name) === -1) {
      lastCol += 1;
      sheet.getRange(1, lastCol).setValue(name);
      headers.push(name);
    }
  });

  var info = getHeaderMap(sheet);
  headers = info.headers;
  lastCol = info.lastCol;
  var map = info.map || {};
  var lastRow = sheet.getLastRow();

  if (lastRow >= 2) {
    var values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();

    function sval(v) { return String(v === undefined || v === null ? "" : v).trim(); }
    function firstLine(v) {
      var parts = sval(v).split(/\r?\n/);
      for (var i = 0; i < parts.length; i++) {
        var x = sval(parts[i]);
        if (x) return x;
      }
      return "Sponsor";
    }
    function yesNo(v) {
      var s = sval(v).toLowerCase();
      if (s === "no" || s === "nej" || s === "false" || s === "0") return "NO";
      return "YES";
    }
    function laneFrom(note) {
      var m = sval(note).match(/(?:ønsket\s*bane|bane\s*sponsor|banesponsor)[^0-9]{0,20}(\d{1,2})/i);
      return m ? Number(m[1]) : "";
    }
    function categoryFrom(row) {
      var note = map.note !== undefined ? sval(row[map.note]) : "";
      var level = map.level !== undefined ? sval(row[map.level]).toLowerCase() : "";
      if (/pakke\s*:\s*guld/i.test(note) || /guld|gold/.test(level)) return "gold";
      if (laneFrom(note)) return "lane";
      if (/pakke\s*:\s*(bronze|sølv|solv)/i.test(note) || /bronze|sølv|solv|støtte|stoette/.test(level)) return "support";
      return "support";
    }

    for (var r = 0; r < values.length; r++) {
      var row = values[r];
      var name = map.name !== undefined ? row[map.name] : "";
      var note = map.note !== undefined ? row[map.note] : "";
      var visible = map.visible !== undefined ? row[map.visible] : "YES";
      var order = map.order !== undefined ? row[map.order] : (r + 1);

      if (map.wall_title !== undefined && !sval(row[map.wall_title])) row[map.wall_title] = firstLine(name);
      if (map.wall_show_extra !== undefined && !sval(row[map.wall_show_extra])) row[map.wall_show_extra] = "YES";
      if (map.wall_kind !== undefined && !sval(row[map.wall_kind])) row[map.wall_kind] = "sponsor";
      if (map.wall_category !== undefined && !sval(row[map.wall_category])) row[map.wall_category] = categoryFrom(row);
      if (map.wall_lane !== undefined && !sval(row[map.wall_lane])) row[map.wall_lane] = laneFrom(note);
      if (map.wall_visible !== undefined && !sval(row[map.wall_visible])) row[map.wall_visible] = yesNo(visible);
      if (map.wall_order !== undefined && !sval(row[map.wall_order])) row[map.wall_order] = order;
    }

    sheet.getRange(2, 1, values.length, lastCol).setValues(values);
  }

  sponsorWallCtaSetup_();
  return { ok: true, tab: "SPONSORER", headers: wanted };
}

  if (lastRow < 2) return { ok: true, updated: 0 };

  if (map.wall_lane === undefined || map.note === undefined) {
    throw new Error("SPONSORER mangler wall_lane eller note");
  }

  var values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();
  var updated = 0;

  function sval(v) { return String(v === undefined || v === null ? "" : v).trim(); }
  function laneFrom(note) {
    var m = sval(note).match(/(?:ønsket\s*bane|bane\s*sponsor|banesponsor)[^0-9]{0,20}(\d{1,2})/i);
    return m ? Number(m[1]) : "";
  }

  for (var r = 0; r < values.length; r++) {
    var lane = laneFrom(values[r][map.note]);
    if (!lane) continue;
    values[r][map.wall_lane] = lane;
    if (map.wall_category !== undefined) values[r][map.wall_category] = "lane";
    updated++;
  }

  sheet.getRange(2, 1, values.length, lastCol).setValues(values);
  SpreadsheetApp.flush();
  return { ok: true, updated: updated };
}

function getSpreadsheet() {
  return SpreadsheetApp.getActiveSpreadsheet();
}

function getSheetByName(tabName) {
  var ss = getSpreadsheet();
  return ss.getSheetByName(tabName);
}

/**
 * Headers + map + last used col
 */
function getHeaderMap(sheet) {
  var lastCol = sheet.getLastColumn();
  var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0] || [];
  var map = {};

  for (var i = 0; i < headers.length; i++) {
    var h = (headers[i] || "").toString().trim();
    headers[i] = h;
    if (h) map[h] = i;
  }

  return { headers: headers, map: map, lastCol: lastCol };
}

/**
 * Læser et ark som liste af objekter med kolonnenavne
 * Row 1 = headers, rækker fra 2 og nedefter = data
 */
function readSheetAsObjects(sheet, limit) {
  var headerInfo = getHeaderMap(sheet);
  var headers = headerInfo.headers;
  var lastCol = headerInfo.lastCol;

  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];

  var rowsToRead = Math.min(limit || 200, lastRow - 1);
  var values = sheet.getRange(2, 1, rowsToRead, lastCol).getValues();

  var out = [];
  for (var r = 0; r < values.length; r++) {
    var rowArr = values[r];
    var obj = {};
    obj._row = r + 2;

    for (var c = 0; c < headers.length; c++) {
      var name = headers[c];
      if (!name) continue;
      obj[name] = rowArr[c];
    }

    out.push(obj);
  }

  return out;
}

/**
 * Opretter ny række i ønsket tab baseret på body
 * Matcher kolonnenavne dynamisk
 */
function createRowFromBody(tabName, body) {
  var ss = getSpreadsheet();
  var sheet = ss.getSheetByName(tabName);
  if (!sheet) throw new Error("Unknown tab: " + tabName);

  var headerInfo = getHeaderMap(sheet);
  var headers = headerInfo.headers;
  var lastCol = headerInfo.lastCol;

  var row = [];
  for (var i = 0; i < lastCol; i++) row.push("");

  var now = new Date();
  var nowStr = formatDate(now);
  var tabKey = String(tabName || "").toUpperCase();

  function setByName(name, value) {
    var idx = headers.indexOf(name);
    if (idx === -1) return;
    row[idx] = value;
  }

  // Standard meta
  setByName("timestamp", nowStr);
  setByName("created_at", nowStr);
  setByName("updated_at", nowStr);

  var statusFieldIndex = headers.indexOf("status");
  if (statusFieldIndex !== -1) {
    row[statusFieldIndex] = (body.status || "").toString().trim() || "pending";
  }

  // Brugervenlig mapping af felter fra body → kolonner
  for (var c = 0; c < headers.length; c++) {
    var hName = headers[c];
    if (!hName) continue;
    if (row[c] !== "") continue;

    var v;

    switch (hName) {
      case "key":
        v = body.row_key || body.key;
        break;

      case "album_key":
        v = body.album_key || body.albumKey || normalizeAlbumKey_(body.category || body.title || body.name);
        break;

      case "slug":
        v = body.slug || body.album_slug || slugifyDa_(body.title || body.album_key || body.albumKey || body.name);
        break;

      case "cover_url":
        v = body.cover_url || body.coverUrl;
        break;

      case "note":
        v = body.note;
        break;

      case "first_name":
        v = body.first_name || body.fornavn || body.firstName;
        break;
      case "last_name":
        v = body.last_name || body.efternavn || body.lastName;
        break;
      case "navn":
      case "name":
        v = body.navn || body.name;
        break;
      case "email":
        v = body.email;
        break;
      case "phone":
      case "telefon":
        v = body.phone || body.telefon;
        break;
      case "address":
      case "adresse":
        v = body.address || body.adresse;
        break;

      case "postcode":
      case "zip":
      case "postnr":
        v = body.postcode || body.postnr || body.zip;
        break;

      case "city":
      case "by":
        v = body.city || body.by;
        break;

      case "package_key":
        v = body.package_key || body.packageKey;
        break;
      case "package_title":
      case "package_id":
      case "pakke":
        v = body.package_title || body.package_id || body.pakke;
        break;

      case "price_amount":
        v = body.price_amount || body.pris;
        break;
      case "price_unit":
        v = body.price_unit;
        break;

      case "level":
      case "niveau":
        v = body.level || body.niveau;
        break;
      case "gender":
      case "køn":
        v = body.gender || body.køn;
        break;

      case "birth_date":
      case "fødselsdato":
        v = body.birth_date || body.birthDate || body.fødselsdato || body.birthdate;
        break;

      case "birth_year":
      case "födselsår":
      case "fødselsår":
        v = body.birth_year || body.birthYear || body.fødselsår;
        if (!v) {
          var bd = body.birth_date || body.birthDate || body.fødselsdato;
          if (bd) {
            var sbd = String(bd);
            var m = sbd.match(/(\d{4})/);
            if (m) v = m[1];
          }
        }
        break;

      case "remark":
      case "bemærkning":
        v = body.remark || body.bemærkning;
        break;

      case "form_version":
        v = body.form_version || body.source_form_version || body.sourceFormVersion || body.source_formVersion;
        break;

      case "source":
        v = body.source || "website";
        break;

      case "payment_method":
        v = body.payment_method || body.method;
        break;

      case "terms_accepted":
        v = body.terms_accepted || body.termsAccepted || body.terms;
        break;
      case "privacy_accepted":
        v = body.privacy_accepted || body.privacyAccepted || body.privacy;
        break;

      case "household_size":
        v = body.household_size || body.household || body.husstand;
        break;
      case "family_members_json":
        if (body.family_members_json) v = body.family_members_json;
        else if (body.family_members) {
          try { v = JSON.stringify(body.family_members); } catch (eJson) { v = ""; }
        } else {
          v = "";
        }
        break;

      default:
        if (body.hasOwnProperty(hName)) {
          v = body[hName];
        }
        break;
    }

    if (v !== undefined && v !== null && v !== "") {
      row[c] = v;
    }
  }

  // GALLERI / ALBUMS default-logik
  if (tabKey === "GALLERI") {
    applyGalleryRowDefaults_(headers, row);
  }

  if (tabKey === "GALLERI_ALBUMS") {
    applyGalleryAlbumDefaults_(headers, row);
  }

  var newRowIndex = sheet.getLastRow() + 1;
  sheet.getRange(newRowIndex, 1, 1, lastCol).setValues([row]);

  var item = {};
  item._row = newRowIndex;
  for (var j = 0; j < lastCol; j++) {
    var name = headers[j];
    if (!name) continue;
    item[name] = row[j];
  }

  return { row: newRowIndex, item: item };
}

function adminUpdateRow(sheet, rowNumber, body) {
  var headerInfo = getHeaderMap(sheet);
  var headers = headerInfo.headers;
  var lastCol = headerInfo.lastCol;

  var rowValues = sheet.getRange(rowNumber, 1, 1, lastCol).getValues()[0];
  var tabKey = String(sheet.getName() || "").toUpperCase();

  for (var c = 0; c < headers.length; c++) {
    var hName = headers[c];
    if (!hName) continue;

    if (body.hasOwnProperty(hName)) {
      rowValues[c] = body[hName];
    }
  }

  // GALLERI / ALBUMS default-logik
  if (tabKey === "GALLERI") {
    applyGalleryRowDefaults_(headers, rowValues);
  }

  if (tabKey === "GALLERI_ALBUMS") {
    applyGalleryAlbumDefaults_(headers, rowValues);
  }

  // auto updated_at hvis kolonnen findes
  var idxUpdated = headers.indexOf("updated_at");
  if (idxUpdated !== -1) {
    rowValues[idxUpdated] = formatDate(new Date());
  }

  sheet.getRange(rowNumber, 1, 1, lastCol).setValues([rowValues]);

  var item = {};
  item._row = rowNumber;
  for (var j = 0; j < lastCol; j++) {
    var name = headers[j];
    if (!name) continue;
    item[name] = rowValues[j];
  }

  return { row: rowNumber, item: item };
}

/**
 * Markerer en indmelding som betalt og opretter medlem i MEDLEMMER.
 * Kopierer værdier baseret på matchende header-navne.
 */
function markPaid(tabName, rowNumber, paymentMethod) {
  var ss = getSpreadsheet();

  // INDMELDINGER
  var inSheet = ss.getSheetByName(tabName);
  if (!inSheet) throw new Error("Unknown INDMELDINGER tab: " + tabName);

  var inHeader = getHeaderMap(inSheet);
  var inHeaders = inHeader.headers;
  var inLastCol = inHeader.lastCol;

  if (rowNumber < 2 || rowNumber > inSheet.getLastRow()) {
    throw new Error("Row out of range in " + tabName + ": " + rowNumber);
  }

  var rowValues = inSheet.getRange(rowNumber, 1, 1, inLastCol).getValues()[0];
  var nowStr = formatDate(new Date());
  var method = (paymentMethod || "MobilePay").toString().trim() || "MobilePay";

  function str__(v) { return (v === undefined || v === null) ? "" : String(v).trim(); }

  function idxIn_(name) { return inHeaders.indexOf(name); }
  function getIn_(name) {
    var idx = idxIn_(name);
    return idx !== -1 ? rowValues[idx] : "";
  }
  function setIn_(name, value) {
    var idx = idxIn_(name);
    if (idx !== -1) rowValues[idx] = value;
  }

  function isBadLikeMobilePay_(v) {
    var s = str__(v).toLowerCase();
    return !s || s === "mobilepay" || s === "mobile pay";
  }

  // Status + betaling
  setIn_("status", "paid");
  setIn_("payment_method", method);
  setIn_("payment_date", nowStr);
  setIn_("updated_at", nowStr);

  // Hvis nogle felter ved en fejl ender med "mobilepay", så rydder vi dem her
  ["member_id", "member_code", "admin_note", "row_id", "household_size", "family_members_json"].forEach(function(k) {
    var idx = idxIn_(k);
    if (idx !== -1 && String(rowValues[idx] || "").toLowerCase() === "mobilepay") {
      rowValues[idx] = "";
    }
  });

  // MEDLEMMER
  var memberSheet = ss.getSheetByName("MEDLEMMER");
  if (!memberSheet) throw new Error("Unknown tab: MEDLEMMER");

  var memHeader = getHeaderMap(memberSheet);
  var memHeaders = memHeader.headers;
  var memLastCol = memHeader.lastCol;

  // Pakkeinfo
  var packageTitle = str__(getIn_("package_title") || getIn_("package") || getIn_("membership"));
  var packageKey = str__(getIn_("package_key"));
  var isFamilyPackage = /familie/i.test(packageTitle) || /familie/i.test(packageKey);

  // Familie-data (fra INDMELDINGER)
  var familyList = [];
  var familyJson = str__(getIn_("family_members_json"));
  if (familyJson) {
    try {
      var parsed = JSON.parse(familyJson);
      if (parsed && parsed.length) familyList = parsed;
    } catch (eFamParse) {}
  }

  var adminNote = str__(getIn_("admin_note"));
  var alreadyCreated = adminNote.indexOf("[FAMILY_CREATED]") !== -1;

  // Vi indsætter altid hovedmedlem som første række, med UPSERT hvis member_id allerede findes
  var baseRowIndex = memberSheet.getLastRow() + 1; // næste tomme række
  var mainRowIndex = baseRowIndex;

  // [HELP:MARKPAID_UPSERT] START
  function findMemberRowById_(id) {
    id = str__(id);
    if (!id) return 0;
    var colIdx = memHeaders.indexOf("member_id");
    if (colIdx === -1) return 0;

    var lastRow = memberSheet.getLastRow();
    if (lastRow < 2) return 0;

    var values = memberSheet.getRange(2, colIdx + 1, lastRow - 1, 1).getValues();
    for (var i = 0; i < values.length; i++) {
      if (str__(values[i][0]) === id) return i + 2;
    }
    return 0;
  }
  // [HELP:MARKPAID_UPSERT] END

  // Sørg for medlemsnummer i INDMELDINGER
  var memberId = str__(getIn_("member_id"));
  if (memberId && /^HDK-\d{4}$/.test(memberId)) {
    var existingRow = findMemberRowById_(memberId);
    if (existingRow) mainRowIndex = existingRow;
  }
  if (!memberId || isBadLikeMobilePay_(memberId) || !/^HDK-\d{4}$/.test(memberId)) {
    memberId = makeMemberId_(mainRowIndex);
    setIn_("member_id", memberId);

    // Edge-case: hvis ID allerede findes, så opdatér den række i stedet
    var existing2 = findMemberRowById_(memberId);
    if (existing2) mainRowIndex = existing2;
  }

  // Sørg for member_code (genbrug eksisterende fra MEDLEMMER hvis den findes)
  var memberCode = str__(getIn_("member_code"));
  var existingCode = "";
  if (mainRowIndex !== baseRowIndex) {
    var idxCode = memHeaders.indexOf("member_code");
    if (idxCode !== -1) existingCode = str__(memberSheet.getRange(mainRowIndex, idxCode + 1).getValue());
  }
  if (existingCode && /^\d{6}$/.test(existingCode)) {
    memberCode = existingCode;
    setIn_("member_code", memberCode);
  } else {
    if (!memberCode || isBadLikeMobilePay_(memberCode) || !/^\d{6}$/.test(memberCode)) {
      memberCode = makeMemberCode_();
      setIn_("member_code", memberCode);
    }
  }

  // Byg række til MEDLEMMER ud fra matchende headers
  function makeMemberRow_(overrides) {
    var row = [];
    for (var i = 0; i < memLastCol; i++) row.push("");

    for (var j = 0; j < memHeaders.length; j++) {
      var hName = memHeaders[j];
      if (!hName) continue;

      var value = "";

      if (hName === "status") {
        value = "active";
      } else if (hName === "payment_method") {
        value = method;
      } else if (hName === "payment_date" || hName === "created_at" || hName === "updated_at" || hName === "start_date") {
        value = nowStr;
      } else if (hName === "source_row_id") {
        var idxRowId = idxIn_("row_id");
        value = (idxRowId !== -1 && rowValues[idxRowId]) ? rowValues[idxRowId] : String(rowNumber);
      } else if (hName === "source_form_version") {
        var idxFV = idxIn_("form_version");
        value = (idxFV !== -1 && rowValues[idxFV]) ? rowValues[idxFV] : "v3";
      } else if (hName === "source") {
        var idxSrc = idxIn_("source");
        value = (idxSrc !== -1 && rowValues[idxSrc]) ? rowValues[idxSrc] : "website";
      } else {
        var idx = idxIn_(hName);
        value = (idx !== -1) ? rowValues[idx] : "";
      }

      row[j] = value;
    }

    // Overrides (styrer fx member_id, navn osv.)
    if (overrides) {
      for (var k in overrides) {
        if (!overrides.hasOwnProperty(k)) continue;
        var idx2 = memHeaders.indexOf(k);
        if (idx2 !== -1) row[idx2] = overrides[k];
      }
    }

    return row;
  }

  var mainMemRow = makeMemberRow_({
    member_id: memberId,
    member_code: memberCode
  });

  // Familie-medlemmer: opret ekstra rækker ved markPaid (MODEL A)
  var familyRows = [];
  var familyCreated = [];
  var familySkipped = "";
  var familyStartRowIndex = 0;

  if (isFamilyPackage && familyList && familyList.length) {
    if (alreadyCreated) {
      familySkipped = "already_created";
    } else {
      // Hvis hovedmedlemmet er UPSERT (findes allerede), så skriv familie nederst for at undgå at overskrive andre rækker
      familyStartRowIndex = (mainRowIndex === baseRowIndex) ? (baseRowIndex + 1) : (memberSheet.getLastRow() + 1);

      // [HELP:MARKPAID_FAMILY_DOB] START
      function pick_(obj, keys) {
        for (var i = 0; i < keys.length; i++) {
          var v = obj[keys[i]];
          if (v !== undefined && v !== null && String(v).trim() !== "") return String(v).trim();
        }
        return "";
      }
      function splitFullName_(full) {
        full = str__(full);
        if (!full) return { fn: "", ln: "" };
        var parts = full.split(/\s+/).filter(Boolean);
        if (parts.length <= 1) return { fn: full, ln: "" };
        var ln = parts.pop();
        var fn = parts.join(" ");
        return { fn: fn, ln: ln };
      }

      for (var f = 0; f < familyList.length; f++) {
        var fam = familyList[f] || {};

        // VIGTIGT: dit sheet bruger typisk first/last (ikke first_name/last_name)
        var fn = pick_(fam, ["first_name", "first", "fornavn", "firstname"]);
        var ln = pick_(fam, ["last_name", "last", "efternavn", "lastname"]);

        if (!fn && !ln) {
          var sp = splitFullName_(pick_(fam, ["name", "full_name", "fullname"]));
          fn = sp.fn; ln = sp.ln;
        }

        var bd = pick_(fam, ["birth_date", "birthDate", "fødselsdato", "birthdate", "dob"]);
        var by = pick_(fam, ["birth_year", "fødselsår", "birthYear"]);
        if (!by && bd) by = String(bd).slice(0, 4);

        var famRowIndex = familyStartRowIndex + f;
        var famMemberId = makeMemberId_(famRowIndex);

        var famRow = makeMemberRow_({
          member_id: famMemberId,
          member_code: "",

          first_name: fn,
          last_name: ln,
          birth_year: by,
          birth_date: bd
        });

        familyRows.push(famRow);
        familyCreated.push({ member_id: famMemberId, first_name: fn, last_name: ln, birth_year: by, birth_date: bd });
      }
      // [HELP:MARKPAID_FAMILY_DOB] END

      if (familyCreated.length) {
        var ids = familyCreated.map(function(x) { return x.member_id; }).join(",");
        var note = adminNote ? (adminNote + " ") : "";
        note += "[FAMILY_CREATED] " + ids;
        setIn_("admin_note", note.trim());
      }
    }
  } else {
    familySkipped = isFamilyPackage ? "no_family_list" : "not_family";
  }

  // Gem opdateringer i INDMELDINGER (inkl. member_id/code + evt. admin_note)
  inSheet.getRange(rowNumber, 1, 1, inLastCol).setValues([rowValues]);

  // [HELP:MARKPAID_WRITE_MEMBER] START
  // Skriv hovedmedlem (UPSERT) + familie-medlemmer (sikkert)
  memberSheet.getRange(mainRowIndex, 1, 1, memLastCol).setValues([mainMemRow]);

  if (familyRows.length) {
    memberSheet.getRange(familyStartRowIndex, 1, familyRows.length, memLastCol).setValues(familyRows);
  }
  // [HELP:MARKPAID_WRITE_MEMBER] END

  return {
    tab: tabName,
    row: rowNumber,
    status: "paid",
    member_row: mainRowIndex,
    member_id: memberId,
    member_code: memberCode,
    family_created: (familyCreated && familyCreated.length) ? true : false,
    family_skipped: familySkipped,
    family_members: familyCreated
  };
}


function rejectRow(tabName, rowNumber) {
  var ss = getSpreadsheet();
  var sheet = ss.getSheetByName(tabName);
  if (!sheet) throw new Error("Unknown tab: " + tabName);

  var headerInfo = getHeaderMap(sheet);
  var headers = headerInfo.headers;
  var lastCol = headerInfo.lastCol;

  if (rowNumber < 2 || rowNumber > sheet.getLastRow()) {
    throw new Error("Row out of range in " + tabName + ": " + rowNumber);
  }

  var rowValues = sheet.getRange(rowNumber, 1, 1, lastCol).getValues()[0];

  var nowStr = formatDate(new Date());

  function setField(name, value) {
    var idx = headers.indexOf(name);
    if (idx !== -1) rowValues[idx] = value;
  }

  setField("status", "rejected");
  setField("updated_at", nowStr);

  sheet.getRange(rowNumber, 1, 1, lastCol).setValues([rowValues]);

  return { tab: tabName, row: rowNumber, status: "rejected" };
}

function normalizeYesNo(value) {
  var v = String(value || "").toLowerCase().trim();
  if (v === "ja" || v === "yes" || v === "true" || v === "1") return true;
  if (v === "nej" || v === "no" || v === "false" || v === "0") return false;
  return value;
}

function normalizeGender(value) {
  var v = String(value || "").toLowerCase().trim();
  if (v === "mand" || v === "m") return "Mand";
  if (v === "kvinde" || v === "k") return "Kvinde";
  if (!v) return "";
  return value;
}

function normalizeLevel(value) {
  var v = String(value || "").toLowerCase().trim();
  if (v === "hygge") return "Hygge";
  if (v === "turnering") return "Turnering";
  if (!v) return "";
  return value;
}

function inSet(tabName, rowNumber, data) {
  var sheet = getSheetByName(tabName);
  if (!sheet) throw new Error("Unknown tab: " + tabName);

  var headerInfo = getHeaderMap(sheet);
  var headers = headerInfo.headers;
  var lastCol = headerInfo.lastCol;

  var rowValues = sheet.getRange(rowNumber, 1, 1, lastCol).getValues()[0];

  for (var c = 0; c < headers.length; c++) {
    var hName = headers[c];
    if (!hName) continue;
    if (data.hasOwnProperty(hName)) {
      rowValues[c] = data[hName];
    }
  }

  var idxUpdated = headers.indexOf("updated_at");
  if (idxUpdated !== -1) rowValues[idxUpdated] = formatDate(new Date());

  sheet.getRange(rowNumber, 1, 1, lastCol).setValues([rowValues]);

  return { tab: tabName, row: rowNumber };
}
